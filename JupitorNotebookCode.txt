import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, classification_report

# 1. LOAD THE PRE-SPLIT DATASETS FROM ONLINE SESSION STORAGE
print("Loading Online Ovarian Cancer Datasets...")
train_df = pd.read_excel("Supplementary data 4.xlsx")
test_df = pd.read_excel("Supplementary data 5.xlsx")

# 2. MEDICAL DATA CLEANING FUNCTION
def clean_medical_data(df):
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = df[col].astype(str).str.replace('"', '', regex=False).str.strip()
            df[col] = pd.to_numeric(df[col], errors='coerce')
    return df

print("Cleaning raw text formats and double quotes...")
train_df = clean_medical_data(train_df)
test_df = clean_medical_data(test_df)

# 3. HANDLING MISSING VALUES USING MEDIAN
train_median = train_df.median(numeric_only=True)
train_df.fillna(train_median, inplace=True)
test_df.fillna(train_median, inplace=True)

# 4. SEPARATE FEATURES & TARGET VARIABLE
X_train = train_df.drop(columns=['TYPE', 'SUBJECT_ID'], errors='ignore')
y_train = train_df['TYPE'].astype(int)

X_test = test_df.drop(columns=['TYPE', 'SUBJECT_ID'], errors='ignore')
y_test = test_df['TYPE'].astype(int)

# 5. BUILD & TRAIN THE RANDOM FOREST CLASSIFIER
print("Training the Ovarian Cancer Prediction Engine (Random Forest)...")
model = RandomForestClassifier(n_estimators=100, random_state=42)
model.fit(X_train, y_train)

# 6. EVALUATE MODEL PERFORMANCE
predictions = model.predict(X_test)
accuracy = accuracy_score(y_test, predictions) * 100

print("\n============================ RESULTS ============================")
print(f"🎉 Model Diagnostic Accuracy Score: {accuracy:.2f}%")
print("=================================================================\n")
print(classification_report(y_test, predictions, target_names=["Malignant (Cancer)", "Benign (Harmless)"]))

# 7. EXPORT PREDICTION RESULTS FOR POWER BI DASHBOARD
output_df = test_df.copy()
output_df['Actual_Status_Code'] = y_test
output_df['Predicted_Status_Code'] = predictions

status_map = {0: "Ovarian Cancer (Malignant)", 1: "Harmless Cyst (Benign)"}
output_df['Actual_Status_Label'] = output_df['Actual_Status_Code'].map(status_map)
output_df['Predicted_Status_Label'] = output_df['Predicted_Status_Code'].map(status_map)

output_filename = "Ovarian_Cancer_PowerBI_Input.csv"
output_df.to_csv(output_filename, index=False)
print(f"📢 Clean dataset successfully exported as '{output_filename}'!")
